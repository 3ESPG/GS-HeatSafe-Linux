# HeatSafe · API central (C# / ASP.NET Core)
# Modelo: os valores entre arrobas são preenchidos pelo provision_app.sh
[Unit]
Description=HeatSafe - API central (C# / ASP.NET Core)
Documentation=file:/opt/heatsafe/infra/README.md
After=network-online.target heatsafe-previsao.service
Wants=network-online.target
# 0 = o systemd nunca "desiste" de reiniciar, por mais falhas que aconteçam
StartLimitIntervalSec=0

[Service]
Type=simple
User=heatsafe
Group=heatsafe
WorkingDirectory=/opt/heatsafe/api
EnvironmentFile=/etc/heatsafe/api.env
Environment=DOTNET_ROOT=/usr/lib/dotnet
# HOME gravável: o ASP.NET guarda ali as chaves de proteção de dados
Environment=HOME=/var/lib/heatsafe
ExecStart=/usr/lib/dotnet/dotnet /opt/heatsafe/api/HeatSafe.Api.dll
# Ganchos de auditoria: registram em /var/log/heatsafe/servicos.log quando sobe/cai e por quê
# ("+" = roda como root; "-" = se o gancho falhar, o serviço não é afetado)
ExecStartPost=-+/opt/heatsafe/infra/scripts/hooks/evento_servico.sh %n iniciado
ExecStopPost=-+/opt/heatsafe/infra/scripts/hooks/evento_servico.sh %n parado

# Tolerância a falhas: qualquer queda (crash, kill -9, OOM) -> reinicia em 3 s
Restart=always
RestartSec=3
KillSignal=SIGINT
TimeoutStopSec=20
SyslogIdentifier=heatsafe-api

# Limites de recursos (cgroups)
MemoryMax=768M
TasksMax=256

# Endurecimento (sandbox do systemd)
NoNewPrivileges=true
PrivateTmp=true
ProtectSystem=full
ProtectHome=true
ProtectKernelTunables=true
ProtectControlGroups=true
ReadWritePaths=/var/lib/heatsafe /var/log/heatsafe

[Install]
WantedBy=multi-user.target

### nginx
$ cat /etc/nginx/sites-available/heatsafe.conf
# =============================================================================
# HeatSafe · nginx — proxy reverso (modelo preenchido pelo provision_app.sh)
# A API e o serviço de previsão escutam só em 127.0.0.1; o mundo externo
# fala apenas com o nginx na porta 80, escolhendo o serviço pelo nome DNS.
# =============================================================================

upstream heatsafe_api      { server 127.0.0.1:5000;      keepalive 16; }
upstream heatsafe_previsao { server 127.0.0.1:8000; keepalive 16; }

# Cabeçalhos comuns de proxy
map $http_upgrade $connection_upgrade { default upgrade; '' ''; }

# ---------- api.heatsafe.local -> API C# ----------
server {
    listen 80;
    server_name api.heatsafe.local;
    access_log /var/log/nginx/heatsafe-api.access.log;
    error_log  /var/log/nginx/heatsafe-api.error.log warn;
    client_max_body_size 5m;

    location / {
        proxy_pass         http://heatsafe_api;
        proxy_http_version 1.1;
        proxy_set_header   Host              $host;
        proxy_set_header   X-Real-IP         $remote_addr;
        proxy_set_header   X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header   X-Forwarded-Proto $scheme;
        proxy_set_header   Upgrade           $http_upgrade;
        proxy_set_header   Connection        $connection_upgrade;
        proxy_read_timeout 60s;
    }
}

# ---------- previsao.heatsafe.local -> serviço de previsão (Java/Spring Boot) ----------
server {
    listen 80;
    server_name previsao.heatsafe.local;
    access_log /var/log/nginx/heatsafe-previsao.access.log;
    error_log  /var/log/nginx/heatsafe-previsao.error.log warn;
    client_max_body_size 50m;   # importação do CSV do INMET

    location / {
        proxy_pass         http://heatsafe_previsao;
        proxy_http_version 1.1;
        proxy_set_header   Host              $host;
        proxy_set_header   X-Real-IP         $remote_addr;
        proxy_set_header   X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header   X-Forwarded-Proto $scheme;
        proxy_set_header   Connection        "";
        proxy_read_timeout 120s;
    }
}

# ---------- heatsafe.local / monitor.heatsafe.local / acesso pelo IP ----------
# Página de status do monitor + caminhos /api/ e /previsao-svc/ para quem não
# consegue usar os nomes DNS (ex.: dispositivo ESP32 acessando pelo IP).
server {
    listen 80 default_server;
    server_name heatsafe.local monitor.heatsafe.local heatsafe-srv _;
    access_log /var/log/nginx/heatsafe-status.access.log;

    root  /var/www/heatsafe-status;
    index index.html;

    location = /status.json {
        add_header Cache-Control "no-store";
        default_type application/json;
        try_files /status.json =404;
    }

    location /api/ {
        proxy_pass       http://heatsafe_api/;
        proxy_set_header Host            $host;
        proxy_set_header X-Real-IP       $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }

    location /previsao-svc/ {
        proxy_pass       http://heatsafe_previsao/;
        proxy_set_header Host            $host;
        proxy_set_header X-Real-IP       $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }

    location / {
        try_files $uri $uri/ =404;
    }
}
